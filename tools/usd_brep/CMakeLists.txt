cmake_minimum_required(VERSION 3.20)
project(viboceros_usd_brep_trial LANGUAGES CXX)

# Standalone native-kernel experiment. No OpenUSD, Python or Packman required.
get_filename_component(USD_BREP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../third_party/usd-brep" ABSOLUTE)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

file(GLOB KERNEL_SOURCES CONFIGURE_DEPENDS "${USD_BREP_ROOT}/source/SMLib/src/*.cpp")
file(GLOB API_SOURCES CONFIGURE_DEPENDS "${USD_BREP_ROOT}/source/SM_API/src/*.cpp")
list(FILTER KERNEL_SOURCES EXCLUDE REGEX "/StdAfx\\.cpp$")
list(FILTER API_SOURCES EXCLUDE REGEX "/StdAfx\\.cpp$")

if(VIBOCEROS_BLEND_RAIL_PATCH)
  find_program(PATCH_EXECUTABLE patch REQUIRED)
  set(BLEND_SOURCE "${USD_BREP_ROOT}/source/SMLib/src/SmBrep.cpp")

  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/usd-brep-patched")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${BLEND_SOURCE}" "${BLEND_PATCH}")
  execute_process(COMMAND "${PATCH_EXECUTABLE}" --batch -o "${BLEND_PATCHED}" "${BLEND_SOURCE}" "${BLEND_PATCH}"
                  RESULT_VARIABLE BLEND_PATCH_STATUS)
  if(NOT BLEND_PATCH_STATUS EQUAL 0)
    message(FATAL_ERROR "Blend rail patch failed; review the pinned upstream source")
  endif()
  list(REMOVE_ITEM KERNEL_SOURCES "${BLEND_SOURCE}")
  list(APPEND KERNEL_SOURCES "${BLEND_PATCHED}")

option(VIBOCEROS_CURVE_FILLET_SIDE_PATCH "Qualify independent curve-filleting side parameters" OFF)
if(VIBOCEROS_CURVE_FILLET_SIDE_PATCH)
  find_program(PATCH_EXECUTABLE patch REQUIRED)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/usd-brep-patched")
  set(CURVE_FILLET_SOURCE "${USD_BREP_ROOT}/source/SMLib/src/SmCurveFilletExecutive.cpp")

  set(CURVE_FILLET_PATCHED "${CMAKE_CURRENT_BINARY_DIR}/usd-brep-patched/SmCurveFilletExecutive.cpp")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CURVE_FILLET_SOURCE}" "${CURVE_FILLET_PATCH}")
  execute_process(COMMAND "${PATCH_EXECUTABLE}" --batch --fuzz=0 -o "${CURVE_FILLET_PATCHED}" "${CURVE_FILLET_SOURCE}" "${CURVE_FILLET_PATCH}"
                  RESULT_VARIABLE CURVE_FILLET_PATCH_STATUS)
  if(NOT CURVE_FILLET_PATCH_STATUS EQUAL 0)
    message(FATAL_ERROR "usd-brep curve fillet side patch failed; review the pinned source")

option(VIBOCEROS_CURVE_FILLET_SEED_PATCH "Qualify explicit fillet seeds and unsnapped contacts" ON)
if(VIBOCEROS_CURVE_FILLET_SEED_PATCH)
  find_program(PATCH_EXECUTABLE patch REQUIRED)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/usd-brep-patched")
  set(FILLET_BASE "${USD_BREP_ROOT}/source/SMLib/src/SmCurveFilletExecutive.cpp")
  if(VIBOCEROS_CURVE_FILLET_SIDE_PATCH)
    set(FILLET_BASE "${CURVE_FILLET_PATCHED}")
  endif()
  set(FILLET_SEED_PATCH "${CMAKE_CURRENT_LIST_DIR}/../../third_party/patches/usd-brep-curve-fillet-seed.patch")
  set(FILLET_SEEDED "${CMAKE_CURRENT_BINARY_DIR}/usd-brep-patched/SmCurveFilletSeeded.cpp")
  set(FILLET_HEADER "${USD_BREP_ROOT}/source/SMLib/inc/SmCurveFilletExecutive.h")
  set(FILLET_HEADER_PATCH "${CMAKE_CURRENT_LIST_DIR}/../../third_party/patches/usd-brep-curve-fillet-seed-header.patch")
  set(FILLET_HEADER_PATCHED "${CMAKE_CURRENT_BINARY_DIR}/usd-brep-patched/SmCurveFilletExecutive.h")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${FILLET_SEED_PATCH}" "${FILLET_HEADER}" "${FILLET_HEADER_PATCH}")
  execute_process(COMMAND "${PATCH_EXECUTABLE}" --batch --fuzz=0 -o "${FILLET_SEEDED}" "${FILLET_BASE}" "${FILLET_SEED_PATCH}" RESULT_VARIABLE FILLET_SEED_STATUS)
  execute_process(COMMAND "${PATCH_EXECUTABLE}" --batch --fuzz=0 -o "${FILLET_HEADER_PATCHED}" "${FILLET_HEADER}" "${FILLET_HEADER_PATCH}" RESULT_VARIABLE FILLET_HEADER_STATUS)
  if(NOT FILLET_SEED_STATUS EQUAL 0 OR NOT FILLET_HEADER_STATUS EQUAL 0)
    message(FATAL_ERROR "usd-brep seeded fillet patch failed; review the pinned source")
  endif()
  list(REMOVE_ITEM KERNEL_SOURCES "${FILLET_BASE}")
  list(APPEND KERNEL_SOURCES "${FILLET_SEEDED}")
endif()
